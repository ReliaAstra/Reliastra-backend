   372→   372→function ResourceDetail({ resource, onClose }: { resource: Resource; onClose: () => void }) {
   373→   373→  const [copiedId, setCopiedId] = useState<string | null>(null);
   374→   374→
   375→   375→  const handleCopy = (text: string, id: string) => {
   376→   376→    navigator.clipboard.writeText(text);
   377→   377→    setCopiedId(id);
   378→   378→    setTimeout(() => setCopiedId(null), 2000);
   379→   379→    toast.success('Copied to clipboard');
   380→   380→  };
   381→   381→
   382→   382→  return (
   383→   383→    <div className="flex h-full flex-col">
   384→   384→      {/* Header */}

   390→   390→            </div>

   394→   394→                <span className="font-mono text-[10px] uppercase tracking-wider text-muted-foreground">
   395→   395→                  {resource.category}
   396→   396→                </span>
   397→   397→                <span className="text-muted-foreground/40">·</span>
   398→   398→                <span className="text-[11px] text-muted-foreground">{resource.readTime}</span>
   399→   399→              </div>
   400→   400→            </div>
   401→   401→          </div>
   402→   402→          <button
   403→   403→            onClick={onClose}
   404→   404→            className="rounded-md p-1.5 text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"
   405→   405→            aria-label="Close"
   406→   406→          >
   407→   407→            <X className="size-4" />
   408→   408→          </button>
   409→   409→        </div>

   411→   411→
   412→   412→      {/* Body - scrollable */}
   413→   413→      <div className="flex-1 overflow-y-auto px-6 py-6">
   414→   414→        <p className="text-sm leading-relaxed text-muted-foreground">

   430→   430→                <div className="space-y-2">
   431→   431→                  {block.content && (
   432→   432→                    <p className="mb-2 text-sm text-muted-foreground">{block.content}</p>
