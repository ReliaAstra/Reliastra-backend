   518→// --- Animated Counter Item ---
   519→function CounterItem({
   520→  target,
   521→  prefix,
   522→  suffix,
   523→  label,
   524→  isZero = false,
   525→}: {
   526→  target: number;
   527→  prefix: string;
   528→  suffix: string;
   529→  label: string;
   530→  isZero?: boolean;
   531→}) {
   532→  const ref = useRef<HTMLDivElement>(null);
   533→  const isInView = useInView(ref, { once: true, margin: '-40px' });
   534→  const count = useMotionValue(0);
   535→  const rounded = useTransform(count, (v) => Math.round(v));
   536→
   537→  useEffect(() => {
   538→    if (!isInView) return;
   539→
   540→    if (isZero) {
   541→      // Subtle scale pulse for $0
   542→      const controls = animate(count, 0, {
   543→        duration: 0.8,
   544→        ease: [0.25, 0.1, 0.25, 1],
   545→      });
   546→      return controls.stop;
   547→    }
   548→
   549→    const controls = animate(count, target, {
   550→      duration: 1.6,
   551→      ease: [0.25, 0.1, 0.25, 1],
   552→    });
   553→    return controls.stop;
   554→  }, [isInView, target, isZero, count]);
   555→
   556→  return (
   557→    <div ref={ref} className="py-8 sm:py-10 text-center px-4 first:pl-0">
   558→      {isZero ? (
   559→        <motion.div
   560→          initial={{ scale: 0.9, opacity: 0 }}
   561→          animate={isInView ? { scale: [1, 1.05, 1], opacity: 1 } : {}}
   562→          transition={{ duration: 0.8, ease: [0.25, 0.1, 0.25, 1] }}
   563→        >
   564→          <span className="font-mono tabular-nums text-3xl font-bold text-foreground sm:text-4xl">
   565→            {prefix}0{suffix}
   566→          </span>
   567→        </motion.div>
   568→      ) : (
   569→        <span className="font-mono tabular-nums text-3xl font-bold text-foreground sm:text-4xl">
   570→          {prefix}<motion.span>{rounded}</motion.span>{suffix}
   571→        </span>
   572→      )}
   573→      <p className="mt-2 font-mono text-xs uppercase tracking-widest text-muted-foreground">
   574→        {label}
   575→      </p>
   576→    </div>
   577→  );
   578→}
