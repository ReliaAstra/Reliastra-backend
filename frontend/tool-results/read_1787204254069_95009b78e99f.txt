   499→// --- Animated Counter Section ---
   500→function CounterSection() {
   501→  const sectionRef = useRef<HTMLElement>(null);
   502→  const [started, setStarted] = useState(false);
   503→
   504→  useEffect(() => {
   505→    const el = sectionRef.current;
   506→    if (!el) return;
   507→
   508→    const observer = new IntersectionObserver(
   509→      ([entry]) => {
   510→        if (entry.isIntersecting) {
   511→          const t = setTimeout(() => setStarted(true), 300);
   512→          observer.disconnect();
   513→        }
   514→      },
   515→      { threshold: 0.1 }
   516→    );
   517→
   518→    observer.observe(el);
   519→    return () => observer.disconnect();
   520→  }, []);
   521→
   522→  return (
   523→    <section ref={sectionRef} className="border-y border-border/40 bg-muted/20">
   524→      <div className="mx-auto max-w-6xl px-4 py-16 sm:px-6 sm:py-20 lg:px-8">
   525→        <motion.div
   526→          initial={{ opacity: 0, y: 12 }}
   527→          whileInView={{ opacity: 1, y: 0 }}
   528→          viewport={{ once: true }}
   529→          transition={{ duration: 0.5, ease: [0.25, 0.1, 0.25, 1] }}
   530→          className="grid grid-cols-2 divide-x divide-border/40 lg:grid-cols-4"
   531→        >
   532→          <CounterItem target={30} prefix="" suffix="%" label="Recurring commission" active={started} />
   533→          <CounterItem target={49} prefix="$" suffix="/mo" label="Starting plan price" active={started} delay={0.1} />
   534→          <CounterItem target={90} prefix="" suffix=" days" label="Attribution window" active={started} delay={0.2} />

   542→// --- Animated Counter Item ---
   543→function CounterItem({
   544→  target,
   545→  prefix,
   546→  suffix,
   547→  label,
   548→  isZero = false,
   549→  active = false,
   550→  delay = 0,
   551→}: {
   552→  target: number;
   553→  prefix: string;
   554→  suffix: string;
   555→  label: string;
   556→  isZero?: boolean;
   557→  active?: boolean;
   558→  delay?: number;
   559→}) {
   560→  const [displayValue, setDisplayValue] = useState(0);
   561→
   562→  useEffect(() => {
   563→    if (!active) return;
   564→
   565→    const startTimeout = setTimeout(() => {
   566→      if (isZero) {
   567→        setDisplayValue(0);
   568→        return;
   569→      }
   570→
   571→      const duration = 1400;
   572→      const startTime = performance.now();
   573→
   574→      const tick = (now: number) => {
   575→        const elapsed = now - startTime;
   576→        const progress = Math.min(elapsed / duration, 1);
   577→        // Ease out cubic
   578→        const eased = 1 - Math.pow(1 - progress, 3);
   579→        setDisplayValue(Math.round(eased * target));
   580→        if (progress < 1) {
   581→          requestAnimationFrame(tick);
   582→        }
   583→      };
   584→
   585→      requestAnimationFrame(tick);
   586→    }, delay * 1000);
   587→
   588→    return () => clearTimeout(startTimeout);
   589→  }, [active, target, isZero, delay]);
   590→
   591→  return (
   592→    <div className="py-8 sm:py-10 text-center px-4 first:pl-0">
   593→      {isZero ? (
   594→        <motion.span
   595→          className="font-mono tabular-nums text-3xl font-bold text-foreground sm:text-4xl"
   596→          initial={{ scale: 0.9, opacity: 0 }}
   597→          animate={active ? { scale: [1, 1.05, 1], opacity: 1 } : {}}
   598→          transition={{ duration: 0.8, ease: [0.25, 0.1, 0.25, 1], delay }}
   599→        >
   600→          {prefix}0{suffix}
   601→        </motion.span>
   602→      ) : (
   603→        <span className="font-mono tabular-nums text-3xl font-bold text-foreground sm:text-4xl">
   604→          {prefix}{displayValue}{suffix}
   605→        </span>
   606→      )}
   607→      <p className="mt-2 font-mono text-xs uppercase tracking-widest text-muted-foreground">
   608→        {label}
   609→      </p>
   610→    </div>
   611→  );
   612→}
